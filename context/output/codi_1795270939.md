```python
#!/usr/bin/env python3
# -*- coding: utf-8 -*-

"""
Programa simple que muestra "Hello, World!" en la salida estándar.

Este script está pensado para ser ejecutado directamente desde la línea de comandos
o importado como módulo sin producir salida inesperada.
"""

def main() -> None:
    """
    Punto de entrada del programa.

    Imprime el mensaje tradicional de prueba en la consola.
    """
    # Mensaje a mostrar
    mensaje = "Hello, World!"
    print(mensaje)


# La condición estándar para que el código solo se ejecute cuando el archivo
# se ejecuta como script y no cuando se importa como módulo.
if __name__ == "__main__":
    main()
```
